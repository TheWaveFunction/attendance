      students.forEach(s => dailyAttendance[s.roll] = false);

    async function addNewStudent() {
      const roll = document.getElementById("newRoll").value.trim();
      const name = document.getElementById("newName").value.trim();

      if (!roll || !name) {
        alert("Please enter Roll Number and Full Name.");
        return;
      }

      try {
        await db.collection("students").doc(roll).set({
          roll: roll,
          name: name
        });

        dailyAttendance[roll] = false;
        document.getElementById("newRoll").value = "";
        document.getElementById("newName").value = "";
      } catch (err) {
        alert("Cloud Error: " + err.message);
      }
    }

    async function deleteStudent(docId, name) {
      if (confirm(`Remove ${name} permanently from Cloud Register?`)) {
        try {
          await db.collection("students").doc(docId).delete();
          delete dailyAttendance[docId];
        } catch (err) {
          alert("Delete Error: " + err.message);
        }
      }
    }

    function generatePDF() {

      const semester = document.getElementById("semester").value;

          1: { cellWidth: 26 },
          3: { cellWidth: 32, fontStyle: 'bold' }
        },

          if (data.column.index === 3) {
            if (data.cell.raw === "PRESENT") {
              data.cell.styles.textColor = [22, 163, 74];
            } else {
              data.cell.styles.textColor = [220, 38, 38];
            }
          }
        }
      });

      doc.save(`Attendance_${semester}_${date}.pdf`);
